{{partial "tk/dbg-template" .}}{{/* tool/include - render another page's content inline .from — string — content path of the page, e.g. /headless/include-example.md A leading /content is stripped, so old-style paths still work (required) The page is found with site.GetPage, so module mounts are honoured and its shortcodes render. params.includeSummary on that page replaces its body. A path that is not a page falls back to readFile, which sees only the real project directory — not mounts. */}} {{- $f := .from -}} {{- if $f -}} {{- $content := "" -}} {{- $p := site.GetPage (replaceRE "^/?content/" "/" $f) -}} {{- with $p -}} {{- $content = .Content -}} {{- with .Params.includeSummary -}} {{- $content = . | markdownify -}} {{- end -}} {{- else -}} {{- if fileExists $f -}} {{- $raw := readFile $f | replaceRE "^---[\\s\\S]+?---" "" -}} {{- if eq ".md" (path.Ext $f) -}} {{- $content = $raw | markdownify -}} {{- else -}} {{- $content = $raw | safeHTML -}} {{- end -}} {{- end -}} {{- end -}} {{- if $content -}} {{- $content -}} {{- else -}} {{- partial "tk/dbg" (dict "err" (printf "%s: from=%q is not a page or a project file" templates.Current.Name $f)) -}} {{- end -}} {{- else -}} {{- partial "tk/dbg" (dict "err" (printf "%s: from is empty" templates.Current.Name)) -}} {{- end -}}
